📊 구글 스프레드시트 Webhook 연동 상태 연동 대기
📖 1분 구글 스프레드시트 연동 가이드 (여기를 클릭하여 펼치기) ▼
  1. 구글 스프레드시트 새 문서 만들기: sheets.new에 접속하여 빈 스프레드시트를 생성합니다 (시트명: 예: 2026 멘탈포럼 문의 접수).
  2. Apps Script 열기: 상단 메뉴에서 확장 프로그램 (Extensions) ➔ Apps Script를 클릭합니다.
  3. 스크립트 코드 붙여넣기: 편집기에 있는 기본 내용을 지우고, 아래의 [코드 복사] 버튼을 눌러 복사한 스크립트를 붙여넣은 뒤 저장(Ctrl+S)합니다.
  4. 웹 앱으로 배포하기: 우측 상단 배포 (Deploy) ➔ 새 배포 (New deployment) 클릭:
    • 유형: 웹 앱 (Web app) 선택 (톱니바퀴 아이콘)
    • 다음 사용자 권한으로 실행 (Execute as): 나 (Me)
    • 액세스 권한이 있는 사용자 (Who has access): 모든 사용자 (Anyone) (매우 중요!)
  5. 웹 앱 URL 복사 & 상단에 붙여넣기: 생성된 https://script.google.com/macros/s/.../exec URL을 복사하여 위의 입력창에 넣고 [URL 저장]을 누르면 즉시 연동 완료됩니다!

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["접수일시 (Timestamp)", "성명 (Name)", "이메일 (Email)", "문의내용 (Message)", "언어 (Language)", "상태 (Status)"]);
      var header = sheet.getRange(1, 1, 1, 6);
      header.setBackground("#1e3a8a").setFontColor("#ffffff").setFontWeight("bold").setHorizontalAlignment("center");
      sheet.setFrozenRows(1);
    }
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch(pErr) { data = e.parameter || {}; }
    } else if (e && e.parameter) {
      data = e.parameter;
    }
    var timestamp = data.timestamp || Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");
    var name = data.name || "미입력";
    var email = data.email || "미입력";
    var message = data.message || "내용 없음";
    var lang = data.lang || "ko";
    sheet.appendRow([timestamp, name, email, message, lang, "접수 완료"]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
